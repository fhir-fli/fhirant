final out = File('tool/review_2026-10-06/results.tsv');
void log(String probe, String caseName, Object result) {
  final line = '$probe\t$caseName\t$result\n';
  out.writeAsStringSync(line, mode: FileMode.append, flush: true);
  stdout.write(line);
}

Future<String> body(Response r) async {
  final b = (await r.readAsString()).replaceAll('\n', ' ');
  return b.length > 160 ? b.substring(0, 160) : b;
}

Future<void> main(List<String> args) async {
  final want = args.toSet();
  bool run(String id) => want.isEmpty || want.contains(id);
  out.writeAsStringSync('probe\tcase\tresult\n', flush: true);
  if (run('P1')) await p1();
  if (run('P2')) await p2();
  if (run('P3')) await p3();
  if (run('P4')) await p4();
  if (run('P5')) p5();

  if (run('P13')) await p13();
  if (run('P14')) await p14();
  if (run('P15')) await p15();
  if (run('P12')) await p12();
  exit(0);
}

// C1: a token verified by signature on a public path is not re-read.
Future<void> p1() async {
  final s = await createTestServer();
  final a = await issueTestToken(s.db, username: 'adminA', role: 'admin', scopes: ['system/*.*']);
  await issueTestToken(s.db, username: 'adminB', role: 'admin', scopes: ['system/*.*']);
  final aId = (await s.db.getUserByUsername('adminA'))!.id;
  String reg(String u) => jsonEncode({'username': u, 'password': 'averylongpassword123', 'role': 'admin'});
  var r = await s.handler(testRequest('POST', '/auth/register', body: reg('ctl1'), authToken: a));
  log('P1', 'control: active admin registers admin', r.statusCode);
  await s.db.deactivateUser(aId);
  await s.db.bumpTokenGeneration(aId);
  r = await s.handler(testRequest('GET', '/Patient', authToken: a));
  log('P1', 'deactivated admin token on GET /Patient', r.statusCode);
  r = await s.handler(testRequest('POST', '/auth/register', body: reg('evil1'), authToken: a));
  log('P1', 'deactivated admin token registers a NEW ADMIN', '${r.statusCode} ${await body(r)}');
  final evil = await s.db.getUserByUsername('evil1');
  log('P1', 'new admin row exists, role', '${evil != null} ${evil?.role}');
  // demoted (role change bumps generation; token still says admin)
  await s.db.activateUser(aId);
  await s.db.updateUserRole(aId, 'readonly');
  await s.db.bumpTokenGeneration(aId);
  r = await s.handler(testRequest('GET', '/Patient', authToken: a));
  log('P1', 'demoted admin stale token on GET /Patient', r.statusCode);
  r = await s.handler(testRequest('POST', '/auth/register', body: reg('evil2'), authToken: a));
  log('P1', 'demoted admin stale token registers a NEW ADMIN', r.statusCode);
  // /auth/password with a deactivated account's token: requires current password -> expect 401
  await s.db.close();
}

// C10: empty encryption key.
Future<void> p7() async {
  log('P7', "encryptionKeyRefusal('', allowPublicKey:false)", encryptionKeyRefusal('', allowPublicKey: false) ?? 'null (no refusal)');
  final dir = Directory('tool/review_2026-10-06/p7').. createSync(recursive: true);
  for (final key in ['', 'a-real-key']) {
    final f = File('${dir.path}/store_${key.isEmpty ? 'empty' : 'real'}.db');
    if (f.existsSync()) f.deleteSync();
    final db = FhirAntDb(NativeDatabase(f, setup: (raw) => applyStoreCipher(raw, key)));
    await db.initialize();
    await db.saveResource(fhir.Patient(id: fhir.FhirString('p1'), name: [fhir.HumanName(family: fhir.FhirString('PLAINTEXTMARKER'))]));
    await db.close();
    final bytes = f.readAsBytesSync();

    final marker = utf8.decode(bytes, allowMalformed: true).contains('PLAINTEXTMARKER');
    log('P7', "key='${key.isEmpty ? '' : 'a-real-key'}' header", '"$head" plaintext marker found: $marker');
  }
}

// C19: a patient-scoped websocket Subscription is pinged by other patients' writes.
Future<void> p11() async {
  final db = FhirAntDb(NativeDatabase.memory());
  await db.initialize();
  final server = FhirAntServer(db, jwtSecret: testJwtSecret);
  await server.startHttp(0);
  final port = server.port!;
  final base = 'http://127.0.0.1:$port';
  final adm = await issueTestToken(db, username: 'adm', role: 'admin', scopes: ['system/*.*']);
  final pat = await issueTestToken(db, username: 'patu', role: 'readonly', scopes: ['patient/*.rs', 'user/Subscription.*'], patientId: 'pat1');
  final client = HttpClient();
  Future<(int, String)> send(String method, String path, String token, [String? bodyStr]) async {
    final req = await client.openUrl(method, Uri.parse('$base$path'));
    req.headers.set('authorization', 'Bearer $token');
    req.headers.set('content-type', 'application/fhir+json');
    if (bodyStr != null) req.write(bodyStr);
    final res = await req.close();
    return (res.statusCode, await res.transform(utf8.decoder).join());
  }
  await send('PUT', '/Patient/pat1', adm, '{"resourceType":"Patient","id":"pat1"}');
  await send('PUT', '/Patient/pat2', adm, '{"resourceType":"Patient","id":"pat2","name":[{"family":"Secret"}]}');
  final sub = await send('POST', '/Subscription', pat, jsonEncode({'resourceType': 'Subscription', 'status': 'requested', 'reason': 'x', 'criteria': 'Observation?subject=Patient/pat2&code=http://loinc.org|718-7', 'channel': {'type': 'websocket'}}));
  log('P11', 'patient token creates websocket Subscription on Patient/pat2 criteria', '${sub.$1} status=${(jsonDecode(sub.$2) as Map)['status']}');
  final subId = (jsonDecode(sub.$2) as Map)['id'];
  final ws = IOWebSocketChannel.connect(Uri.parse('ws://127.0.0.1:$port/ws'), headers: {'authorization': 'Bearer $pat'});
  final messages = <String>[];
  final sub2 = ws.stream.listen((m) => messages.add('$m'));
  ws.sink.add('bind $subId');
  await Future<void>.delayed(const Duration(milliseconds: 300));
  log('P11', 'bind reply', messages.join('|'));
  messages.clear();
  final w = await send('POST', '/Observation', adm, jsonEncode({'resourceType': 'Observation', 'status': 'final', 'code': {'coding': [{'system': 'http://loinc.org', 'code': '718-7'}]}, 'subject': {'reference': 'Patient/pat2'}}));
  await Future<void>.delayed(const Duration(milliseconds: 800));
  log('P11', "admin writes Patient/pat2's Observation (status $w); patient pat1's socket received", messages.isEmpty ? 'nothing' : messages.join('|'));
  final rd = await send('GET', '/Observation?subject=Patient/pat2', pat);
  log('P11', 'control: patient token GET /Observation?subject=Patient/pat2 total', '${rd.$1} total=${(jsonDecode(rd.$2) as Map)['total']}');
  await sub2.cancel();
  await ws.sink.close();
  client.close(force: true);
  await server.stop();
  await db.close();
}

// C21: translation/parse on the server isolate, outside the deadline.
Future<void> p13() async {
  for (final n in [500, 2000, 8000]) {
    final sb = StringBuffer('library L version \'1\'\nusing FHIR version \'4.0.1\'\ncontext Patient\n');
    for (var i = 0; i < n; i++) {
      sb.writeln('define D$i: ${i} + ${i} * 2 - 1');
    }
    final cql = sb.toString();
    final sw = Stopwatch()..start();
    try {
      libraryFromCql(cql);
    } catch (e) {
      log('P13', 'cql defines=$n', 'threw $e');
      continue;
    }
    log('P13', 'cql defines=$n bytes=${cql.length} translate ms', sw.elapsedMilliseconds);
  }
  final engine = await FHIRPathEngine.create(WorkerContext());
  for (final n in [1000, 10000, 100000]) {
    final expr = List.filled(n, '1').join(' + ');
    final sw = Stopwatch()..start();
    try {
      engine.parse(expr);
      log('P13', 'fhirpath parse terms=$n ms', sw.elapsedMilliseconds);
    } catch (e) {
      log('P13', 'fhirpath parse terms=$n', 'threw ${e.runtimeType} after ${sw.elapsedMilliseconds} ms');
    }
  }
  final nested = '${'('.padLeft(5000, '(')}1${')'.padLeft(5000, ')')}';
  final sw = Stopwatch()..start();
  try {
    engine.parse(nested);
    log('P13', 'fhirpath parse 5000 nested parens ms', sw.elapsedMilliseconds);
  } catch (e) {
    log('P13', 'fhirpath parse 5000 nested parens', 'threw ${e.runtimeType} after ${sw.elapsedMilliseconds} ms');
  }
}

class CountingCache extends DbResourceCache {
  CountingCache(super.db);
  int names = 0;
  int namesMs = 0;
  @override
  Future<List<String>> getResourceNames() async {
    final sw = Stopwatch()..start();
    final r = await super.getResourceNames();
    names++;
    namesMs += sw.elapsedMilliseconds;
    return r;
  }
}

import 'dart:io';
import 'package:drift/native.dart';
import 'package:fhirant_db/fhirant_db.dart';
import 'package:sqlite3/sqlite3.dart' show sqlite3;
import 'package:test/test.dart';
void main() {
  test('what a FhirAntDb backup is', () async {
    final dir = Directory.systemTemp.createTempSync('r1_probe_');
    final out = File('tool/review_2026-09-17/fix_r1/00c_fhirantdb_backup_format.log').openSync(mode: FileMode.write);
    void log(String s) { out.writeStringSync('$s\n'); out.flushSync(); print(s); }
    final prod = FhirAntDb(NativeDatabase(File('${dir.path}/prod.db'), setup: (raw) => applyStoreCipher(raw, 'k')));
    await prod.initialize();
    final plain = FhirAntDb(NativeDatabase.memory());
    await plain.initialize();
    await prod.copyEncrypted('${dir.path}/from_prod.sqlite', 'pw');
    await plain.copyEncrypted('${dir.path}/from_plain.sqlite', 'pw');
    for (final n in ['cipher', 'legacy', 'legacy_page_size', 'kdf_iter', 'hmac_use', 'plaintext_header_size', 'page_size']) {
      final a = (await prod.customSelect('PRAGMA $n').get()).map((r) => r.data).toList();
      final b = (await plain.customSelect('PRAGMA $n').get()).map((r) => r.data).toList();
      log('PRAGMA $n: prod=$a plain=$b');
    }
    await prod.close(); await plain.close();
    for (final f in ['prod.db', 'from_prod.sqlite', 'from_plain.sqlite']) {
      final head = File('${dir.path}/$f').openSync().readSync(24);
      final res = <String>[];
      for (final legacy in [4, 0]) {
        final r = sqlite3.open('${dir.path}/$f')..execute("PRAGMA cipher='sqlcipher'")..execute('PRAGMA legacy=$legacy')..execute("PRAGMA key='${f == 'prod.db' ? 'k' : 'pw'}'");
        try { r.select('SELECT count(*) FROM sqlite_master'); res.add('legacy=$legacy opens'); } catch (e) { res.add('legacy=$legacy FAILS'); }
        r.close();
      }
      final r = sqlite3.open('${dir.path}/$f')..execute("PRAGMA cipher='sqlcipher'")..execute("PRAGMA key='${f == 'prod.db' ? 'k' : 'pw'}'");
      try { r.select('SELECT count(*) FROM sqlite_master'); res.add('no-legacy-pragma opens'); } catch (e) { res.add('no-legacy-pragma FAILS'); }
      r.close();
      log('$f size=${File('${dir.path}/$f').lengthSync()} bytes16-23=${head.sublist(16)} ${res.join(' | ')}');
    }
    out.closeSync();
    dir.deleteSync(recursive: true);
  });
}
